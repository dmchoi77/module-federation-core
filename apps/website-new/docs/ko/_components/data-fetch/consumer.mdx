::: info Note
현재는 [Modern.js](/integrations/framework/modernjs/index.html)만 SSR 환경의 data fetching을 지원합니다.
:::

Consumer에서는 [createLazyComponent](/guide/bridge/react/load-component.html#what-is-createlazycomponent) API를 사용해 remote 컴포넌트를 로드하고 데이터를 가져와야 합니다.

```tsx
import { getInstance } from '@module-federation/enhanced/runtime';
import { ERROR_TYPE } from '@module-federation/bridge-react/data-fetch';
import { lazyLoadComponentPlugin } from '@module-federation/bridge-react/data-fetch';

const instance = getInstance();
instance.registerPlugins([lazyLoadComponentPlugin()]);

const List = instance.createLazyComponent({
  loader: () => import('remote/List'),
  loading: '로딩 중...',
  export: 'default',
  fallback: ({ error, errorType, dataFetchMapKey }) => {
    console.error(error);
    if (errorType === ERROR_TYPE.LOAD_REMOTE) {
      return <div>remote 로드 실패</div>;
    }
    if (errorType === ERROR_TYPE.DATA_FETCH) {
      return <div>data fetch 실패: {dataFetchMapKey}</div>;
    }
    return <div>알 수 없는 오류 타입입니다</div>;
  },
});

export default function Index() {
  return <List />;
}
```
